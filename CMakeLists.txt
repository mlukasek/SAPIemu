cmake_minimum_required(VERSION 3.24)
project(SAPIemu VERSION 0.7.0 LANGUAGES C CXX)
set(SAPIEMU_VERSION_SUFFIX "-beta")    # release stage, shown with the version

# MSVC: the C runtime linked in (no Visual C++ redistributable needed on other PCs);
# set before any target, SDL3 from FetchContent follows it
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SAPIEMU_BUILD_GUI   "Build the SDL3/ImGui frontend" ON)
option(SAPIEMU_BUILD_TESTS "Build unit and integration tests" ON)
option(SAPIEMU_SOURCE_FALLBACK "Look for the data in the source tree as the last resort (not for packages)" ON)

set(TP ${CMAKE_CURRENT_SOURCE_DIR}/third_party)

if(MSVC)
    add_compile_options(/utf-8 /W3 /permissive- /Zc:__cplusplus)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# --- third party -------------------------------------------------------------

# floooh/chips Z80 core (zlib license), implementation compiled as C
add_library(chips_z80 STATIC src/cpu/z80_impl.c)
target_include_directories(chips_z80 PUBLIC ${TP}/chips)

# superzazu/8080 Intel 8080 core (MIT license), compiled as C
add_library(i8080 STATIC ${TP}/i8080/i8080.c)
target_include_directories(i8080 PUBLIC ${TP}/i8080)

# nukeykt/Nuked-OPL3 YMF262 core (LGPL-2.1+), used for the YM3812 of MPH-1V
add_library(nuked_opl3 STATIC ${TP}/nuked-opl3/opl3.c)
target_include_directories(nuked_opl3 PUBLIC ${TP}/nuked-opl3)
if(MSVC)
    target_compile_options(nuked_opl3 PRIVATE /wd4244)
endif()

add_library(nlohmann_json INTERFACE)
target_include_directories(nlohmann_json INTERFACE ${TP}/json)

add_library(httplib INTERFACE)
target_include_directories(httplib INTERFACE ${TP}/httplib)
if(WIN32)
    target_link_libraries(httplib INTERFACE ws2_32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(httplib INTERFACE Threads::Threads)
endif()

# --- emulator core (no GUI dependency) ----------------------------------------

file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS
    src/core/*.cpp src/cpu/*.cpp src/chips/*.cpp src/cards/*.cpp
    src/input/*.cpp src/serial/*.cpp src/mcp/*.cpp src/plant/*.cpp)
add_library(sapicore STATIC ${CORE_SOURCES})
target_include_directories(sapicore PUBLIC src)
target_include_directories(sapicore PRIVATE ${TP}/minimp3)
target_link_libraries(sapicore PUBLIC chips_z80 i8080 nuked_opl3 nlohmann_json httplib)
if(NOT WIN32)
    target_link_libraries(sapicore PUBLIC Threads::Threads)
endif()
target_compile_definitions(sapicore PUBLIC
    SAPIEMU_VERSION="${PROJECT_VERSION}${SAPIEMU_VERSION_SUFFIX}")
# The source tree as the last place to look for the data (roms/, machines/): for builds outside
# the tree (a local preset on C:). Off in the release presets: a packaged exe must never fall back
# to the author's working copy (tools/package.py checks it).
if(SAPIEMU_SOURCE_FALLBACK)
    target_compile_definitions(sapicore PUBLIC SAPIEMU_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
endif()

# --- frontend -------------------------------------------------------------------

if(SAPIEMU_BUILD_GUI)
    # Prefer an installed SDL3 (Linux distro packages), otherwise fetch the release.
    find_package(SDL3 CONFIG QUIET)
    if(NOT SDL3_FOUND)
        include(FetchContent)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SDL3
            URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(SDL3)
    endif()

    add_library(imgui STATIC
        ${TP}/imgui/imgui.cpp ${TP}/imgui/imgui_draw.cpp ${TP}/imgui/imgui_tables.cpp
        ${TP}/imgui/imgui_widgets.cpp ${TP}/imgui/imgui_demo.cpp
        ${TP}/imgui/backends/imgui_impl_sdl3.cpp ${TP}/imgui/backends/imgui_impl_sdlrenderer3.cpp)
    target_include_directories(imgui PUBLIC ${TP}/imgui ${TP}/imgui/backends)
    target_link_libraries(imgui PUBLIC SDL3::SDL3)

    file(GLOB_RECURSE FRONTEND_SOURCES CONFIGURE_DEPENDS src/frontend/*.cpp)
    # GUI application (no console window on Windows)
    add_executable(sapiemu WIN32 src/main.cpp ${FRONTEND_SOURCES})
    target_link_libraries(sapiemu PRIVATE sapicore imgui SDL3::SDL3)
    target_compile_definitions(sapiemu PRIVATE SAPIEMU_GUI=1)
    # a Visual Studio solution generated by CMake (-G "Visual Studio 17 2022") starts the GUI, not ALL_BUILD
    set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT sapiemu)
endif()

# Console tool: headless runs, scripting, tests
add_executable(sapiemu-cli src/main.cpp)
target_link_libraries(sapiemu-cli PRIVATE sapicore)

# Windows: the icon and the version of both exe files (Explorer, file properties)
if(WIN32)
    set(SAPIEMU_ICON "${CMAKE_CURRENT_SOURCE_DIR}/src/res/sapiemu.ico")
    set(SAPIEMU_FULL_VERSION "${PROJECT_VERSION}${SAPIEMU_VERSION_SUFFIX}")
    set(SAPIEMU_EXES sapiemu-cli)
    if(SAPIEMU_BUILD_GUI)
        list(APPEND SAPIEMU_EXES sapiemu)
    endif()
    foreach(exe ${SAPIEMU_EXES})
        set(SAPIEMU_EXE "${exe}.exe")
        configure_file(src/res/sapiemu.rc.in ${CMAKE_CURRENT_BINARY_DIR}/${exe}.rc @ONLY)
        target_sources(${exe} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${exe}.rc)
        # a new icon (tools/make_icon.py) rebuilds the resources
        set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/${exe}.rc PROPERTIES OBJECT_DEPENDS ${SAPIEMU_ICON})
    endforeach()
endif()

# Runs CP/M programs on the PC (M80/L80 for the SAPI system sources)
add_executable(cpmrun tools/cpmrun.cpp)
target_link_libraries(cpmrun PRIVATE chips_z80)

if(SAPIEMU_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
