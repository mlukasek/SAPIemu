# Each test is a small standalone executable returning 0 on success.
# Tests that need ROMs/media find the repository through the data root lookup.

function(sapi_test name)
    add_executable(${name} ${name}.cpp test_util.h)
    target_link_libraries(${name} PRIVATE sapicore)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# Intel 8080 core: TST8080, 8080PRE and CPUTEST are quick, 8080EXM takes minutes (slow, release only)
add_executable(test_i8080 test_i8080.cpp test_util.h)
target_link_libraries(test_i8080 PRIVATE sapicore)
add_test(NAME test_i8080 COMMAND test_i8080)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_test(NAME test_8080exm COMMAND test_i8080 8080EXM.COM)
    set_tests_properties(test_8080exm PROPERTIES LABELS slow TIMEOUT 1800)
    # the 8085 mode against the CRCs of a real Intel D8085AH-1
    add_test(NAME test_8085exm COMMAND test_i8080 8085EXM.COM 8085)
    set_tests_properties(test_8085exm PROPERTIES LABELS slow TIMEOUT 1800)
endif()
